Markov Model k-Fold Cross-Validation Workflow
==============================================

Goal
-----
Evaluate a Markov model (order k) using k-fold cross-validation
and report ROC / AUC / Precision-Recall curves.

Libraries
----------
- sklearn.model_selection → StratifiedKFold
- sklearn.metrics → roc_curve, auc, precision_recall_curve
- numpy
- matplotlib

Workflow
---------

1. For each k (model order):
   - Loop over dict_of_dfs.
Done
2. Prepare data:
   - X = sequences
   - y = bound/unbound labels

4. For each fold:
   - Split into train / validation
   - Train Markov model on training sequences only
   - Score validation sequences (log-likelihood)
   - Store:
       • true labels
       • predicted scores

5. After all folds:
   - Concatenate all validation predictions
   - Compute:
       • ROC curve
       • AUC
       • Precision–Recall curve

6. Plot curves:
   - ROC: FPR vs TPR (+ diagonal baseline)
   - PR: Recall vs Precision
   - Compare across different k values if desired

Key Rules
----------
- Never train on validation fold.
- Always pool predictions from all folds before computing AUC.\





def train_markov_models(train_df, which_factor, order_k):

    bound_seqs = train_df.loc[
        train_df[which_factor] == 1, "sequence"
    ].tolist()

    unbound_seqs = train_df.loc[
        train_df[which_factor] == 0, "sequence"
    ].tolist()

    bdd_model = markov_k(bound_seqs, order_k)
    unb_model = markov_k(unbound_seqs, order_k)

    return bdd_model, unb_model


def compute_llr(df, bdd_model, unb_model):

    ll_bdd = df["sequence"].apply(
        lambda s: sequence_score(s, bdd_model)
    )

    ll_unb = df["sequence"].apply(
        lambda s: sequence_score(s, unb_model)
    )

    return ll_bdd - ll_unb
def cross_validate_markov(df, which_factor, order_k=2,
                          n_splits=5, random_state=42):

    kf = KFold(
        n_splits=n_splits,
        shuffle=True,
        random_state=random_state
    )

    all_fold_results = []
    aucs = []

    mean_fpr = np.linspace(0, 1, 100)
    tprs = []

    for fold, (train_idx, test_idx) in enumerate(kf.split(df)):

        train_df = df.iloc[train_idx].reset_index(drop=True)
        test_df = df.iloc[test_idx].reset_index(drop=True)

        # ---- Train on training fold only ----
        bdd_model, unb_model = train_markov_models(
            train_df,
            which_factor,
            order_k
        )

        # ---- Score test fold ----
        test_llr = compute_llr(test_df, bdd_model, unb_model)
        y_true = test_df[which_factor].values

        # ---- Metrics ----
        auc = roc_auc_score(y_true, test_llr)
        aucs.append(auc)

        fpr, tpr, _ = roc_curve(y_true, test_llr)

        interp_tpr = np.interp(mean_fpr, fpr, tpr)
        interp_tpr[0] = 0.0
        tprs.append(interp_tpr)

        print(f"Fold {fold+1} AUC: {auc:.4f}")

        # Store fold predictions
        fold_df = test_df.copy()
        fold_df["llr"] = test_llr
        fold_df["fold"] = fold + 1
        all_fold_results.append(fold_df)

    results_df = pd.concat(all_fold_results, ignore_index=True)

    print("\nMean AUC:", np.mean(aucs))
    print("Std AUC :", np.std(aucs))

    return results_df, aucs, mean_fpr, tprs

def plot_mean_roc(mean_fpr, tprs, aucs, title="ROC Curve"):

    mean_tpr = np.mean(tprs, axis=0)
    mean_tpr[-1] = 1.0
    mean_auc = np.mean(aucs)

    plt.figure()
    plt.plot(mean_fpr, mean_tpr,
             label=f"Mean ROC (AUC = {mean_auc:.3f})")

    plt.plot([0,1], [0,1], linestyle="--")
    plt.xlabel("False Positive Rate")
    plt.ylabel("True Positive Rate")
    plt.title(title)
    plt.legend()
    plt.show()

from sklearn.metrics import (
    roc_auc_score,
    roc_curve,
    precision_recall_curve,
    average_precision_score
)

def cross_validate_markov(df, which_factor, order_k=2,
                          n_splits=5, random_state=42):

    kf = KFold(
        n_splits=n_splits,
        shuffle=True,
        random_state=random_state
    )

    all_fold_results = []

    aucs = []
    aps = []

    mean_fpr = np.linspace(0, 1, 100)
    mean_recall = np.linspace(0, 1, 100)

    roc_tprs = []
    pr_precisions = []

    for fold, (train_idx, test_idx) in enumerate(kf.split(df)):

        train_df = df.iloc[train_idx].reset_index(drop=True)
        test_df = df.iloc[test_idx].reset_index(drop=True)

        # ---- Train ----
        bdd_model, unb_model = train_markov_models(
            train_df,
            which_factor,
            order_k
        )

        # ---- Score ----
        test_llr = compute_llr(test_df, bdd_model, unb_model)
        y_true = test_df[which_factor].values

        # ---- ROC ----
        auc = roc_auc_score(y_true, test_llr)
        aucs.append(auc)

        fpr, tpr, _ = roc_curve(y_true, test_llr)
        interp_tpr = np.interp(mean_fpr, fpr, tpr)
        interp_tpr[0] = 0.0
        roc_tprs.append(interp_tpr)

        # ---- PRC ----
        precision, recall, _ = precision_recall_curve(y_true, test_llr)
        ap = average_precision_score(y_true, test_llr)
        aps.append(ap)

        # Interpolate precision at common recall grid
        interp_precision = np.interp(mean_recall, recall[::-1], precision[::-1])
        pr_precisions.append(interp_precision)

        print(f"Fold {fold+1} AUC: {auc:.4f} | AP: {ap:.4f}")

        fold_df = test_df.copy()
        fold_df["llr"] = test_llr
        fold_df["fold"] = fold + 1
        all_fold_results.append(fold_df)

    results_df = pd.concat(all_fold_results, ignore_index=True)

    print("\nMean ROC AUC:", np.mean(aucs))
    print("Std ROC AUC :", np.std(aucs))
    print("Mean AP     :", np.mean(aps))
    print("Std AP      :", np.std(aps))

    return (results_df,
            aucs, roc_tprs, mean_fpr,
            aps, pr_precisions, mean_recall)

    mean_precision = np.mean(pr_precisions, axis=0)
    mean_ap = np.mean(aps)

    plt.figure()
    plt.plot(mean_recall, mean_precision,

    plt.xlabel("Recall")

    plt.show()
